\question[TDM21]{1} Hỏi hàm nào dưới đây không phải là một nguyên hàm của hàm số $f(x)=2x-4$?
\fourchoices{$x^2-4x+1$}{$2x^2-4x$}{$(x-2)^2-4$}{$(x-4)^2+4x$}

\question[TDM21]{2} $F(x)$ là một nguyên hàm của hàm số $f(x)$ trên $K$ nếu
\twochoices{$F(x)=-f(x)$ với $\forall x\in K$.}{$F(x)=[f(x)]^2$ với $\forall x\in K$.}
{$f'(x)=F(x)$ với $\forall x\in K$.}{$F'(x)=f(x)$ với $\forall x\in K$.}

\question[TDM21]{3} Chọn phát biểu sai:
\twochoices{$\displaystyle\int\!\bigl(f(x)+g(x)\bigr)\,dx=\int f(x)\,dx+\int g(x)\,dx$.}
{$\displaystyle\int\!\bigl(f(x)-g(x)\bigr)\,dx=\int f(x)\,dx-\int g(x)\,dx$.}
{$\displaystyle\int dx=x+C$.}
{$\displaystyle\int k\,f(x)\,dx=k\int f(x)\,dx$ với $\forall k\in\mathbb R$.}

\question[TDM21]{4} Họ nguyên hàm của hàm số $f(x)=3x^2-4+\dfrac{1}{x}$ tương ứng là:
\twochoices{$x^3-4x-\dfrac{1}{x^2}+C$}{$3x^3-4x+\ln|x|+C$}
{$x^3-4x+\ln|x|+C$}{$x^3-4+\ln|x|+C$}

\question[TDM21]{5} Họ nguyên hàm của hàm số $f(x)=2\sin x-3$ tương ứng là:
\fourchoices{$-2\cos x-3x+C$}{$2\cos x-3x+C$}{$-2\cos x+3x+C$}{$2\cos x-3+C$}

\question[TDM21]{6} Họ nguyên hàm của hàm số $f(x)=3\cos x-\dfrac{1}{\cos^2x}$ tương ứng là:
\fourchoices{$3\sin x+\tan x+C$}{$3\sin x-\tan x+C$}{$-3\sin x-\tan x+C$}{$3\sin x+\cot x+C$}

\question[TDM21]{7} Họ nguyên hàm của hàm số $f(x)=\dfrac{2}{\sin^2x}-x^5$ tương ứng là:
\fourchoices{$-2\cot x-\dfrac{x^6}{6}+C$}{$\cot x-\dfrac{x^6}{6}+C$}{$-\cot x-x^6+C$}{$2\tan x-\dfrac{x^6}{6}+C$}

\question[TDM21]{8} Họ nguyên hàm của hàm số $f(x)=\dfrac{1}{\sqrt{x}}$ tương ứng là:
\fourchoices{$\sqrt{x}+C$}{$2x\sqrt{x}+C$}{$x^{\frac12}+C$}{$2\sqrt{x}+C$}

\question[TDM21]{9} Hãy tính $\displaystyle\int\!\left(\sin x-\dfrac{2}{x}\right)\,dx$?
\twochoices{$\cos x-2\ln|x|+C$}{$-\cos x-\ln|x|+C$}
{$-\cos x-2\ln|x|+C$}{$-\cos x-2\ln x+C$}

\question[TDM21]{10} Hãy tính $\displaystyle\int\!\bigl(e^x+2^{x+1}\bigr)\,dx$?
\twochoices{$e^x+2^{x+1}+C$}{$e^x+\dfrac{2^{x+1}}{\ln2}+C$}
{$e^x-2^x+C$}{$e^x+2^x+C$}

\question[TDM31]{11} Hãy tính $\displaystyle\int\!\bigl(x^2+1\bigr)\bigl(x^3+x\bigr)\,dx$?
\twochoices{$F(x)=\dfrac{x^6}{6}+\dfrac{x^4}{4}+\dfrac{x^2}{2}+C$}
{$F(x)=\dfrac{x^6}{3}-\dfrac{x^4}{4}+\dfrac{x^2}{2}+C$}
{$F(x)=\dfrac{x^6}{6}+\dfrac{x^4}{2}+\dfrac{x^2}{2}+C$}
{$F(x)=\dfrac{x^6}{3}+\dfrac{x^3}{3}+\dfrac{x^2}{4}+C$}

\question[TDM31]{12} Hãy tính $\displaystyle\int\!\dfrac{(x^3+1)^2}{x^4}\,dx$?
\twochoices{$F(x)=\dfrac{x^3}{6}+2\ln|x|-\dfrac{1}{3x^3}+C$}
{$F(x)=\dfrac{x^3}{4}+2\ln|x|-\dfrac{1}{x^3}+C$}
{$F(x)=\dfrac{x^3}{3}+2\ln|x|+\dfrac{1}{3x^3}+C$}
{$F(x)=\dfrac{x^3}{3}+2\ln|x|-\dfrac{1}{3x^3}+C$}

\question[TDM32]{13} Từ độ cao $20\,\mathrm{m}$ một quả bóng được ném thẳng đứng lên trên với vận tốc được tính bởi công thức $v=-10t+20$ ($\mathrm{m/s}$).\par
Hỏi trong các mệnh đề dưới đây, mệnh đề nào là \textbf{đúng}, mệnh đề nào là \textbf{sai}?
\tfbox{Ở thời điểm ban đầu vật có vận tốc là $20\,\mathrm{m/s}$.}
{Độ cao của vật ở thời điểm $t$ là $s(t)=-5t^2+20t+20$ ($\mathrm{m}$).}
{Ở thời điểm vật có vận tốc là $-20\,\mathrm{m/s}$ thì nó có độ cao là $20\,\mathrm{m}$.}
{Ở thời điểm $t_2=5$ giây thì vật chạm đất.}

\question[TDM32]{14} Cho $F(x)$ và $2F(x)-x^2$ là nguyên hàm của hàm số $f(x)$. Biết rằng $F(1)=5$.\par
Hỏi trong các mệnh đề dưới đây, mệnh đề nào là \textbf{đúng}, mệnh đề nào là \textbf{sai}?
\tfbox{$f(x)=F'(x)$.}
{$f(x)=2F'(x)-2x$.}
{$f(x)=4x^2$.}
{$F(2)=8$.}

\question[TDM32]{15} Biết tốc độ thay đổi lượng khách tham dự một cuộc triển lãm được biểu diễn bằng hàm số $B'(t)=at+b$ (khách/giờ). Trong đó $t$ tính bằng giờ, bắt đầu mở cửa thì coi như đã có $200$ khách đứng đợi sẵn và tham dự luôn, sau khi mở cửa được $8$ giờ thì lượng khách là đông nhất, sau đúng $20$ giờ thì cuộc triển lãm kết thúc và không còn khách nào.\par
Hỏi trong các mệnh đề dưới đây, mệnh đề nào là \textbf{đúng}, mệnh đề nào là \textbf{sai}?
\tfbox{Số lượng khách ban đầu là $B(0)=200$.}
{Số lượng khách ở thời điểm $t$ là $B(t)=-5t^2+40t+200$ (khách).}
{Số lượng khách tham gia đông nhất là $360$ khách. (Số khách có mặt ở một thời điểm)}
{Nếu trong $4$ giờ đầu có $300$ lượt khách ra về thì tổng số lượt khách tham dự trong $4$ giờ đầu là $620$ (khách).}

\question[TDM31]{16} Cho hàm số $f(x)$ có đạo hàm xác định trên toàn $\mathbb R$. Biết $(x^2-x)$ là một nguyên hàm của hàm số $f(x)e^x$. Hỏi trong các mệnh đề dưới đây, mệnh đề nào là \textbf{đúng}, mệnh đề nào là \textbf{sai}?
\tfbox{$f(x)=(2x-1)e^{-x}$.}
{$f'(x)=e^x(2x-3)$.}
{$\displaystyle\int f'(x)e^x\,dx=-x^2+3x+C,\ C\in\mathbb R$.}
{Gọi $F(x)$ là một nguyên hàm của $\dfrac{[f'(x)e^x]^2}{x}$, có $F(1)=0$. Khi đó $F(2)=24$.}
